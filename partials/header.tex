% Apply link: \jdapply{mailto:...}{label}
\newcommand{\jdapply}[2]{%
  \href{#1}{\tcbox[on line, colback=jdprimary, colframe=jdprimary, arc=1.5mm,
    left=2.5mm, right=2.5mm, top=1mm, bottom=1mm, boxsep=0pt]{%
    \sffamily\small\bfseries\color{white}#2\enspace$\rightarrow$}}}

}
